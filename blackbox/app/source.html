    Le même script dans TradingView

    Cette page calcule les mêmes indicateurs que ton script CARO. Pour les alertes dans TradingView, colle-le dans le Pine Editor, puis Enregistrer et Ajouter au graphique.

    //@version=6
// CARO · Indicateurs Black Box
// Tendance (EMA 20/50/200), volatilité (Bollinger), momentum (RSI, MACD),
// signaux de croisement, tableau de bord et alertes. Outil d'analyse, pas un conseil financier.

indicator("CARO · Indicateurs Black Box", shorttitle="CARO", overlay=true)

// ---------- Réglages ----------
grpT = "Tendance"
emaFastLen = input.int(20, "EMA rapide", minval=1, group=grpT)
emaMidLen = input.int(50, "EMA moyenne", minval=1, group=grpT)
emaSlowLen = input.int(200, "EMA lente", minval=1, group=grpT)
showBg = input.bool(true, "Colorer le fond selon la tendance", group=grpT)

grpB = "Bandes de Bollinger"
showBB = input.bool(true, "Afficher les bandes", group=grpB)
bbLen = input.int(20, "Longueur", minval=1, group=grpB)
bbMult = input.float(2.0, "Écart-type", minval=0.1, step=0.1, group=grpB)

grpM = "Momentum"
rsiLen = input.int(14, "RSI : longueur", minval=2, group=grpM)
rsiHi = input.int(70, "RSI : zone surachetée", minval=50, maxval=100, group=grpM)
rsiLo = input.int(30, "RSI : zone survendue", minval=0, maxval=50, group=grpM)
macdFast = input.int(12, "MACD : rapide", minval=1, group=grpM)
macdSlow = input.int(26, "MACD : lente", minval=1, group=grpM)
macdSig = input.int(9, "MACD : signal", minval=1, group=grpM)

grpS = "Affichage"
showSignals = input.bool(true, "Flèches de croisement", group=grpS)
showRsiMarks = input.bool(true, "Points RSI extrêmes", group=grpS)
showTable = input.bool(true, "Tableau de bord", group=grpS)

// ---------- Calculs ----------
emaFast = ta.ema(close, emaFastLen)
emaMid = ta.ema(close, emaMidLen)
emaSlow = ta.ema(close, emaSlowLen)
[bbMid, bbUp, bbLow] = ta.bb(close, bbLen, bbMult)
rsi = ta.rsi(close, rsiLen)
[macdLine, sigLine, macdHist] = ta.macd(close, macdFast, macdSlow, macdSig)

trendUp = close > emaSlow and emaFast > emaMid
trendDown = close < emaSlow and emaFast < emaMid

// Croisement EMA rapide/moyenne, confirmé par le RSI et le MACD
bullSig = ta.crossover(emaFast, emaMid) and rsi > 50 and macdLine > sigLine
bearSig = ta.crossunder(emaFast, emaMid) and rsi < 50 and macdLine < sigLine
rsiOverbought = ta.crossover(rsi, rsiHi)
rsiOversold = ta.crossunder(rsi, rsiLo)

// ---------- Graphique ----------
plot(emaFast, "EMA rapide", color=color.new(#4fd1ff, 0), linewidth=1)
plot(emaMid, "EMA moyenne", color=color.new(#b58cff, 0), linewidth=1)
plot(emaSlow, "EMA lente", color=color.new(#ffb84d, 0), linewidth=2)

pUp = plot(showBB ? bbUp : na, "Bollinger haute", color=color.new(#8796b8, 40))
pLow = plot(showBB ? bbLow : na, "Bollinger basse", color=color.new(#8796b8, 40))
fill(pUp, pLow, color=color.new(#8796b8, 92), title="Zone Bollinger")

bgcolor(showBg and trendUp ? color.new(#3df5a8, 93) : showBg and trendDown ? color.new(#ff4d6d, 93) : na, title="Fond tendance")

plotshape(showSignals and bullSig, "Croisement haussier", shape.triangleup, location.belowbar, color.new(#3df5a8, 0), size=size.small, text="HAUSSE")
plotshape(showSignals and bearSig, "Croisement baissier", shape.triangledown, location.abovebar, color.new(#ff4d6d, 0), size=size.small, text="BAISSE")
plotshape(showRsiMarks and rsiOverbought, "RSI suracheté", shape.circle, location.abovebar, color.new(#ffb84d, 0), size=size.tiny)
plotshape(showRsiMarks and rsiOversold, "RSI survendu", shape.circle, location.belowbar, color.new(#4fd1ff, 0), size=size.tiny)

// ---------- Tableau de bord ----------
var table dash = table.new(position.top_right, 2, 5, bgcolor=color.new(#06080d, 10), border_color=#1c2740, border_width=1, frame_color=#1c2740, frame_width=1)

if showTable and barstate.islast
    txt = #e4ecff
    dist = (close - emaSlow) / emaSlow * 100
    table.cell(dash, 0, 0, "CARO", text_color=#d8f36a, text_size=size.small)
    table.cell(dash, 1, 0, syminfo.ticker, text_color=txt, text_size=size.small)
    table.cell(dash, 0, 1, "Tendance", text_color=txt, text_size=size.small)
    table.cell(dash, 1, 1, trendUp ? "Haussière" : trendDown ? "Baissière" : "Neutre", text_color=trendUp ? #3df5a8 : trendDown ? #ff4d6d : #8796b8, text_size=size.small)
    table.cell(dash, 0, 2, "RSI " + str.tostring(rsiLen), text_color=txt, text_size=size.small)
    table.cell(dash, 1, 2, str.tostring(rsi, "#.0"), text_color=rsi > rsiHi ? #ffb84d : rsi < rsiLo ? #4fd1ff : txt, text_size=size.small)
    table.cell(dash, 0, 3, "MACD", text_color=txt, text_size=size.small)
    table.cell(dash, 1, 3, macdLine > sigLine ? "Au-dessus du signal" : "Sous le signal", text_color=macdLine > sigLine ? #3df5a8 : #ff4d6d, text_size=size.small)
    table.cell(dash, 0, 4, "Écart EMA " + str.tostring(emaSlowLen), text_color=txt, text_size=size.small)
    table.cell(dash, 1, 4, str.tostring(dist, "+#.##;-#.##") + " %", text_color=dist >= 0 ? #3df5a8 : #ff4d6d, text_size=size.small)

// ---------- Alertes ----------
// Le message peut être remplacé plus tard par le JSON d'un Signal Bot 3Commas.
alertcondition(bullSig, "CARO · Croisement haussier", "CARO : croisement haussier EMA {{ticker}} à {{close}}")
alertcondition(bearSig, "CARO · Croisement baissier", "CARO : croisement baissier EMA {{ticker}} à {{close}}")
alertcondition(rsiOverbought, "CARO · RSI suracheté", "CARO : RSI suracheté sur {{ticker}} à {{close}}")
alertcondition(rsiOversold, "CARO · RSI survendu", "CARO : RSI survendu sur {{ticker}} à {{close}}")

    Les signaux décrivent ce que le prix a fait, pas ce qu'il va faire. Ce n'est pas un conseil financier.
